function doPost(e){ const ss=SpreadsheetApp.getActiveSpreadsheet(); const sh=ss.getSheetByName('dados')||ss.insertSheet('dados'); const body=JSON.parse(e.postData.contents); const out=ContentService.createTextOutput(); out.setMimeType(ContentService.MimeType.JSON); if(body.action==='save'){ // Salvar db em chunks de 45.000 chars (limite seguro por célula) const dbStr=body.db||'[]'; const CHUNK=45000; const nChunks=Math.ceil(dbStr.length/CHUNK); sh.getRange('A1').setValue(nChunks); for(let i=0;i<nChunks;i++){ sh.getRange('A'+(i+2)).setValue(dbStr.slice(i*CHUNK,(i+1)*CHUNK)); } // Limpar chunks antigos se o novo for menor try{const old=parseInt(sh.getRange('A1').getValue())||0; for(let i=nChunks;i<old;i++)sh.getRange('A'+(i+2)).setValue(''); }catch(ex){} sh.getRange('B1').setValue(body.caps||'{}'); out.setContent(JSON.stringify({ok:true})); }else if(body.action==='load'){ // Ler db em chunks let nChunks=1; try{nChunks=parseInt(sh.getRange('A1').getValue())||1;}catch(ex){} // Compatibilidade: se A1 contém JSON (versão antiga), ler diretamente let db=''; const a1=sh.getRange('A1').getValue()||''; if(a1.startsWith('[')||a1.startsWith('{')){ db=a1; // formato antigo } else { nChunks=parseInt(a1)||1; for(let i=0;i<nChunks;i++){ db+=(sh.getRange('A'+(i+2)).getValue()||''); } } if(!db||db==='0')db='[]'; const caps=sh.getRange('B1').getValue()||'{}'; out.setContent(JSON.stringify({db,caps})); }else if(body.action==='setPwd'){ sh.getRange('C1').setValue(body.pwd||''); out.setContent(JSON.stringify({ok:true})); }else if(body.action==='getPwd'){ const pwd=sh.getRange('C1').getValue()||''; out.setContent(JSON.stringify({pwd})); }else if(body.action==='sendResetCode'){ // Verifica se o e-mail é autorizado (salvo em D1, separados por vírgula) const authEmails=(sh.getRange('D1').getValue()||'').split(',').map(s=>s.trim().toLowerCase()); const email=(body.email||'').toLowerCase(); if(!authEmails.includes(email)){ out.setContent(JSON.stringify({ok:false,error:'E-mail não autorizado para redefinição de senha.'})); return out; } // Gera código OTP de 6 dígitos e expira em 10 min const code=String(Math.floor(100000+Math.random()*900000)); const exp=new Date().getTime()+10*60*1000; sh.getRange('E1').setValue(code); sh.getRange('F1').setValue(exp); sh.getRange('G1').setValue(email); // Envia e-mail MailApp.sendEmail({ to: email, subject: '[VEP-TJRO] Código de redefinição de senha', body: 'Código de verificação: ' + code + '\n\n' + 'Válido por 10 minutos. Se não foi você quem solicitou, ignore este e-mail.\n\n' + 'Sistema VEP — Controle Prisional · TJ/RO' }); out.setContent(JSON.stringify({ok:true})); }else if(body.action==='verifyResetCode'){ const code=sh.getRange('E1').getValue(); const exp=sh.getRange('F1').getValue(); const storedEmail=sh.getRange('G1').getValue(); const now=new Date().getTime(); if(String(code)===String(body.code) && now<Number(exp) && storedEmail===(body.email||'')){ // Marca como verificado (H1) sem apagar ainda — setPwdVerified vai usar sh.getRange('H1').setValue('verified:'+body.email); out.setContent(JSON.stringify({ok:true})); } else { out.setContent(JSON.stringify({ok:false,error:'Código incorreto ou expirado.'})); } }else if(body.action==='setPwdVerified'){ // Só permite trocar senha se H1 tiver o token verificado para o mesmo e-mail const token=sh.getRange('H1').getValue(); if(token==='verified:'+(body.email||'')){ sh.getRange('C1').setValue(body.pwd||''); // Limpa tokens de reset sh.getRange('E1').setValue(''); sh.getRange('F1').setValue(''); sh.getRange('G1').setValue(''); sh.getRange('H1').setValue(''); out.setContent(JSON.stringify({ok:true})); } else { out.setContent(JSON.stringify({ok:false,error:'Sessão de verificação inválida ou expirada. Reinicie o processo.'})); } } return out; }
